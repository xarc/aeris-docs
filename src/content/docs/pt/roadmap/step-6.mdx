## Objetivo

Programar um jogo da cobra completo.

## Instruções

1. Abra o simulador de linguagem RISC-V e ative o painel gráfico pelo botão do menu de opções.
2. Ative o **Modo Interativo**, o alternador no cabeçalho do painel.
3. No editor de texto do simulador, transcreva o programa RISC-V abaixo, monte e execute.

:::note
Abra o painel gráfico **antes** de montar o código, porque alterná-lo depois da montagem reinicia a simulação.
:::

## Pseudocódigo

O pseudocódigo abaixo é a referência de alto nível para o programa RISC-V da seção seguinte.

```
VARIAVEIS
    endereco_base, endereco_teclado, cor_cobra, cor_comida, tecla: INTEIRO
    segmento_x, segmento_y: VETOR DE INTEIRO
    comprimento, capacidade, direcao: INTEIRO
    comida, candidato, indice, posicao: INTEIRO
    i, nova_x, nova_y, colidiu, cresceu, limite: INTEIRO
    endereco: INTEIRO
    memoria: VETOR DE INTEIRO

INICIO

endereco_base <- 0xFF000000
endereco_teclado <- 0xFF010000
cor_cobra <- 0x0000F000
cor_comida <- 0x00FF0000

comprimento <- 4
capacidade <- 576     # painel inteiro: 32 colunas e 18 linhas
direcao <- 1          # 0=cima, 1=direita, 2=baixo, 3=esquerda

segmento_x[0] <- 10
segmento_x[1] <- 9
segmento_x[2] <- 8
segmento_x[3] <- 7
segmento_y[0] <- 9
segmento_y[1] <- 9
segmento_y[2] <- 9
segmento_y[3] <- 9

comida <- 180         # indice do pixel: linha 5, coluna 20
candidato <- 180

i <- 0
ENQUANTO i < comprimento FACA
    endereco <- segmento_y[i] * 32
    endereco <- endereco + segmento_x[i]
    endereco <- endereco * 4
    endereco <- endereco + endereco_base
    memoria[endereco] <- cor_cobra
    i <- i + 1
FIM-ENQUANTO

endereco <- comida * 4
endereco <- endereco + endereco_base
memoria[endereco] <- cor_comida

ENQUANTO VERDADEIRO FACA
    ESPERAR 150     # pausa de 150 milissegundos

    tecla <- memoria[endereco_teclado]

    SE tecla != 0 ENTAO
        SE tecla == 38 E direcao != 2 ENTAO
            direcao <- 0
        FIM-SE
        SE tecla == 40 E direcao != 0 ENTAO
            direcao <- 2
        FIM-SE
        SE tecla == 37 E direcao != 1 ENTAO
            direcao <- 3
        FIM-SE
        SE tecla == 39 E direcao != 3 ENTAO
            direcao <- 1
        FIM-SE
        memoria[endereco_teclado] <- 0
    FIM-SE

    nova_x <- segmento_x[0]
    nova_y <- segmento_y[0]

    SE direcao == 0 ENTAO
        nova_y <- nova_y - 1
        SE nova_y < 0 ENTAO
            nova_y <- 17
        FIM-SE
    FIM-SE

    SE direcao == 2 ENTAO
        nova_y <- nova_y + 1
        SE nova_y >= 18 ENTAO
            nova_y <- 0
        FIM-SE
    FIM-SE

    SE direcao == 3 ENTAO
        nova_x <- nova_x - 1
        SE nova_x < 0 ENTAO
            nova_x <- 31
        FIM-SE
    FIM-SE

    SE direcao == 1 ENTAO
        nova_x <- nova_x + 1
        SE nova_x >= 32 ENTAO
            nova_x <- 0
        FIM-SE
    FIM-SE

    posicao <- nova_y * 32
    posicao <- posicao + nova_x

    cresceu <- 0
    SE posicao == comida ENTAO
        comprimento <- comprimento + 1
        cresceu <- 1
    FIM-SE

    limite <- comprimento - 1

    colidiu <- 0
    i <- 0
    ENQUANTO i < limite E colidiu == 0 FACA
        SE nova_x == segmento_x[i] E nova_y == segmento_y[i] ENTAO
            colidiu <- 1
        FIM-SE
        i <- i + 1
    FIM-ENQUANTO

    SE colidiu == 1 ENTAO
        ENCERRAR
    FIM-SE

    SE cresceu == 0 ENTAO
        endereco <- segmento_y[comprimento - 1] * 32
        endereco <- endereco + segmento_x[comprimento - 1]
        endereco <- endereco * 4
        endereco <- endereco + endereco_base
        memoria[endereco] <- 0
    FIM-SE

    i <- comprimento - 1
    ENQUANTO i > 0 FACA
        segmento_x[i] <- segmento_x[i - 1]
        segmento_y[i] <- segmento_y[i - 1]
        i <- i - 1
    FIM-ENQUANTO

    segmento_x[0] <- nova_x
    segmento_y[0] <- nova_y

    endereco <- segmento_y[0] * 32
    endereco <- endereco + segmento_x[0]
    endereco <- endereco * 4
    endereco <- endereco + endereco_base
    memoria[endereco] <- cor_cobra

    SE cresceu == 1 ENTAO
        SE comprimento == capacidade ENTAO
            ENCERRAR
        FIM-SE

        candidato <- candidato + 37
        SE candidato >= 576 ENTAO
            candidato <- candidato - 576
        FIM-SE

        indice <- candidato
        endereco <- indice * 4
        endereco <- endereco + endereco_base

        ENQUANTO memoria[endereco] != 0 FACA
            indice <- indice + 1
            SE indice >= 576 ENTAO
                indice <- 0
            FIM-SE
            endereco <- indice * 4
            endereco <- endereco + endereco_base
        FIM-ENQUANTO

        comida <- indice
        memoria[endereco] <- cor_comida
    FIM-SE
FIM-ENQUANTO

FIM
```

## Programa RISC-V

```asm
.text
main:
    lui  t0, 0xFF000              # framebuffer base = 0xFF000000
    lui  t3, 0xFF010              # registrador de teclado = 0xFF010000

    lui  s2, 0x10010              # s2 = 0x10010000 (base do vetor de colunas)
    lui  s3, 0x10011              # s3 = 0x10011000 (base do vetor de linhas)

    addi s4, zero, 4              # comprimento inicial
    addi s6, zero, 576            # capacidade = 32 * 18 (painel inteiro)
    addi s1, zero, 1              # direcao inicial: direita

    addi s7, zero, 180            # indice da comida (linha 5, coluna 20)
    addi s9, zero, 180            # indice candidato do proximo sorteio

    # corpo inicial: colunas 10, 9, 8 e 7, todas na linha 9
    addi t1, zero, 10
    sw   t1, 0(s2)
    addi t1, zero, 9
    sw   t1, 4(s2)
    addi t1, zero, 8
    sw   t1, 8(s2)
    addi t1, zero, 7
    sw   t1, 12(s2)

    addi t1, zero, 9
    sw   t1, 0(s3)
    sw   t1, 4(s3)
    sw   t1, 8(s3)
    sw   t1, 12(s3)

    addi s5, zero, 0
init_draw:
    slli t6, s5, 2
    add  t2, s2, t6
    lw   a4, 0(t2)
    add  t2, s3, t6
    lw   a5, 0(t2)
    slli t2, a5, 5
    add  t2, t2, a4
    slli t2, t2, 2
    add  t2, t2, t0
    lui  t1, 0xF                  # verde, 0x0000F000
    sw   t1, 0(t2)
    addi s5, s5, 1
    blt  s5, s4, init_draw

    slli t2, s7, 2                # endereco da comida inicial
    add  t2, t2, t0
    lui  t1, 0xFF0                # vermelho (comida)
    sw   t1, 0(t2)

tick:
    addi a0, zero, 150            # pausa de 150 ms por volta do jogo
    addi a7, zero, 32             # chamada de sistema 32 = pausa (a0 = ms)
    ecall

    lw   t4, 0(t3)
    beq  t4, zero, move_step

    addi t5, zero, 38
    beq  t4, t5, check_up
    addi t5, zero, 40
    beq  t4, t5, check_down
    addi t5, zero, 37
    beq  t4, t5, check_left
    addi t5, zero, 39
    beq  t4, t5, check_right
    jal  zero, consume_key

check_up:
    addi t5, zero, 2              # oposto de cima (0) e baixo (2)
    beq  s1, t5, consume_key      # indo para baixo? ignora a tecla
    addi s1, zero, 0
    jal  zero, consume_key

check_down:
    addi t5, zero, 0              # oposto de baixo (2) e cima (0)
    beq  s1, t5, consume_key
    addi s1, zero, 2
    jal  zero, consume_key

check_left:
    addi t5, zero, 1              # oposto de esquerda (3) e direita (1)
    beq  s1, t5, consume_key
    addi s1, zero, 3
    jal  zero, consume_key

check_right:
    addi t5, zero, 3              # oposto de direita (1) e esquerda (3)
    beq  s1, t5, consume_key
    addi s1, zero, 1

consume_key:
    sw   x0, 0(t3)

move_step:
    lw   a4, 0(s2)
    lw   a5, 0(s3)

    addi t5, zero, 0
    beq  s1, t5, move_up
    addi t5, zero, 1
    beq  s1, t5, move_right
    addi t5, zero, 2
    beq  s1, t5, move_down
    jal  zero, move_left

move_up:
    addi a0, a5, -1
    bge  a0, zero, up_ok
    addi a0, zero, 17
up_ok:
    addi a1, a0, 0
    addi a0, a4, 0
    jal  zero, head_computed

move_down:
    addi a1, a5, 1
    addi t5, zero, 18
    blt  a1, t5, down_ok
    addi a1, zero, 0
down_ok:
    addi a0, a4, 0
    jal  zero, head_computed

move_left:
    addi a0, a4, -1
    bge  a0, zero, left_ok
    addi a0, zero, 31
left_ok:
    addi a1, a5, 0
    jal  zero, head_computed

move_right:
    addi a0, a4, 1
    addi t5, zero, 32
    blt  a0, t5, right_ok
    addi a0, zero, 0
right_ok:
    addi a1, a5, 0

head_computed:
    slli t5, a1, 5
    add  t5, t5, a0               # t5 = indice da nova cabeca (y * 32 + x)
    addi a7, zero, 0              # a7 = 1 se a cobra cresce nesta volta
    bne  t5, s7, skip_grow
    addi s4, s4, 1
    addi a7, zero, 1
skip_grow:

    # segmentos a verificar: comprimento - 1. Se cresceu, o ultimo indice
    # ainda nao tem segmento. Se nao cresceu, a cauda libera a celula
    addi t6, s4, -1

    addi s5, zero, 0
collision_loop:
    bge  s5, t6, no_collision
    slli t5, s5, 2
    add  t2, s2, t5
    lw   a2, 0(t2)
    add  t2, s3, t5
    lw   a3, 0(t2)
    bne  a0, a2, collision_next
    bne  a1, a3, collision_next
    jal  zero, fim                # colisao com o proprio corpo: encerra
collision_next:
    addi s5, s5, 1
    jal  zero, collision_loop

no_collision:
    bne  a7, zero, do_shift
    addi t6, s4, -1
    slli t6, t6, 2
    add  t2, s2, t6
    lw   a2, 0(t2)
    add  t2, s3, t6
    lw   a3, 0(t2)

do_shift:
    addi s5, s4, -1
shift_loop:
    beq  s5, zero, shift_done
    slli t6, s5, 2
    addi t5, t6, -4
    add  t2, s2, t5
    lw   a4, 0(t2)
    add  t2, s3, t5
    lw   a5, 0(t2)
    add  t2, s2, t6
    sw   a4, 0(t2)
    add  t2, s3, t6
    sw   a5, 0(t2)
    addi s5, s5, -1
    jal  zero, shift_loop
shift_done:
    sw   a0, 0(s2)
    sw   a1, 0(s3)

    bne  a7, zero, draw_new_head
    slli t2, a3, 5
    add  t2, t2, a2
    slli t2, t2, 2
    add  t2, t2, t0
    sw   x0, 0(t2)

draw_new_head:
    slli t2, a1, 5
    add  t2, t2, a0
    slli t2, t2, 2
    add  t2, t2, t0
    lui  t1, 0xF                  # verde, 0x0000F000
    sw   t1, 0(t2)

    beq  a7, zero, tick

spawn_food:
    beq  s4, s6, fim              # painel inteiro ocupado: vitoria, encerra

    addi s9, s9, 37               # passo 37: coprimo com 576, percorre todo o painel
    addi t5, zero, 576
    blt  s9, t5, candidate_ok
    addi s9, s9, -576             # passou do ultimo pixel, volta ao inicio
candidate_ok:
    addi t6, s9, 0

find_free:
    slli t2, t6, 2
    add  t2, t2, t0               # endereco do pixel candidato
    lw   t5, 0(t2)
    beq  t5, zero, free_found     # pixel apagado: livre
    addi t6, t6, 1
    addi t5, zero, 576
    blt  t6, t5, find_free
    addi t6, zero, 0              # passou do ultimo pixel, volta ao primeiro
    jal  zero, find_free

free_found:
    addi s7, t6, 0                # novo indice da comida
    lui  t1, 0xFF0                # vermelho (comida)
    sw   t1, 0(t2)
    jal  zero, tick

fim:
    addi a7, zero, 10             # chamada de sistema 10 = encerra o programa
    ecall
```

:::note
Este programa utiliza a instrução `ecall`. As chamadas de sistema suportadas pelo simulador estão listadas em [Syscalls](/aeris-docs/pt/riscv/syscalls/).
:::

## Explicação

O corpo é guardado em dois vetores, um com as colunas e outro com as linhas dos segmentos, e o índice 0 corresponde à cabeça. A cada iteração, o programa faz uma pausa, calcula a nova posição da cabeça e verifica se ela coincide com a comida, caso em que o corpo cresce, ou com algum segmento do corpo, caso em que o programa termina. Quando a cobra não cresce, a cauda é desconsiderada nessa comparação, pois libera sua célula. O corpo é deslocado da cauda para a cabeça, com cada segmento assumindo a posição do anterior. Os píxeis são numerados de 0 a 575, de modo que o píxel da coluna x e da linha y recebe o número y * 32 + x. A posição da comida parte de um contador que avança 37 posições por vez, em contagem circular, e, se o píxel candidato estiver aceso, o programa examina os seguintes até achar um píxel apagado. A tecla correspondente ao sentido oposto ao da direção atual é ignorada. A partida termina por colisão ou quando o corpo ocupa todos os píxeis, e o programa se encerra com [`ecall`](/aeris-docs/pt/riscv/syscalls/).
