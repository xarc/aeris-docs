## Objective

Make the line move continuously across the panel, without user intervention.

## Instructions

1. Open the RISC-V simulator and turn on the graphic panel through the button in the options menu.
2. Transcribe the RISC-V program below into the editor, assemble it and run it.

:::note
Open the graphic panel **before** assembling the code, because toggling it after assembling resets the simulation.
:::

## Pseudocode

The pseudocode below is the high-level reference for the RISC-V program in the next section.

```
VARIABLES
    y, row_base, base_address, color, x, limit, address: INTEGER
    head, tail: INTEGER
    memory: ARRAY OF INTEGER

BEGIN

y <- 9
row_base <- y * 32
base_address <- 0xFF000000
color <- 0x0000F000

x <- 13
limit <- 19

WHILE x < limit DO
    address <- row_base + x
    address <- address * 4
    address <- address + base_address
    memory[address] <- color
    x <- x + 1
END-WHILE

head <- 18
tail <- 13

WHILE TRUE DO
    address <- row_base + tail
    address <- address * 4
    address <- address + base_address
    memory[address] <- 0

    tail <- tail + 1
    IF tail >= 32 THEN
        tail <- 0
    END-IF

    head <- head + 1
    IF head >= 32 THEN
        head <- 0
    END-IF

    address <- row_base + head
    address <- address * 4
    address <- address + base_address
    memory[address] <- color

    WAIT 100    
END-WHILE

END
```

## RISC-V Program

```asm
.text
main:
    addi t0, zero, 9                # y = 9 (center row of the grid)
    slli t1, t0, 5                  # t1 = y * 32 (row_base)
    lui  t2, 0xFF000                # t2 = 0xFF000000 (base_address)
    lui  t3, 0xF                    # t3 = 0x0000F000 (green)

    # initial drawing of the line: columns 13 to 18
    addi t4, zero, 13
    addi a0, zero, 19
draw_initial:
    add  t6, t1, t4
    slli t6, t6, 2
    add  t6, t6, t2
    sw   t3, 0(t6)
    addi t4, t4, 1
    blt  t4, a0, draw_initial

    addi t4, t4, -1                 # head = 18
    addi t5, zero, 13               # tail = 13
    addi a1, zero, 32               # column limit, used in the comparisons

main_loop:
    # erases the tail pixel
    add  t6, t1, t5
    slli t6, t6, 2
    add  t6, t6, t2
    sw   zero, 0(t6)

    # advances the tail, going back to 0 after column 31
    addi t5, t5, 1
    blt  t5, a1, cauda_ok
    addi t5, zero, 0
cauda_ok:

    # advances the head, going back to 0 after column 31
    addi t4, t4, 1
    blt  t4, a1, cabeca_ok
    addi t4, zero, 0
cabeca_ok:

    # lights the new head pixel
    add  t6, t1, t4
    slli t6, t6, 2
    add  t6, t6, t2
    sw   t3, 0(t6)

    # pause to make the movement visible
    addi a7, zero, 32               # system call: pause
    addi a0, zero, 100              # time in milliseconds
    ecall

    jal  zero, main_loop
```

:::note
This program uses the `ecall` instruction. The system calls supported by the simulator are listed in [Syscalls](/aeris-docs/riscv/syscalls/).
:::

## Explanation

The movement uses two position references, the head and the tail, which advance together on each iteration: the tail erases the pixel that stops being part of the line, and the head lights the pixel the line advances into. On passing the last column, the position goes back to the first one. The initial drawing is now done by a loop, instead of repeated blocks of code. The pause between the iterations is a system call, with the code 32 in `a7` and the time in milliseconds in `a0`. Since the pause counts real time, the movement stays visible regardless of the execution frequency configured in the simulator.
