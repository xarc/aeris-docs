## Objective

Program a complete Snake game.

## Instructions

1. Open the RISC-V simulator and turn on the graphic panel through the button in the options menu.
2. Turn on **Interactive Mode**, the toggle in the panel header.
3. Transcribe the RISC-V program below into the editor, assemble it and run it.

:::note
Open the graphic panel **before** assembling the code, because toggling it after assembling resets the simulation.
:::

## Pseudocode

The pseudocode below is the high-level reference for the RISC-V program in the next section.

```
VARIABLES
    base_address, keyboard_address, snake_color, food_color, key: INTEGER
    segment_x, segment_y: ARRAY OF INTEGER
    length, capacity, direction: INTEGER
    food, candidate, index, position: INTEGER
    i, new_x, new_y, collided, grew, limit: INTEGER
    address: INTEGER
    memory: ARRAY OF INTEGER

BEGIN

base_address <- 0xFF000000
keyboard_address <- 0xFF010000
snake_color <- 0x0000F000
food_color <- 0x00FF0000

length <- 4
capacity <- 576     # whole panel: 32 columns and 18 rows
direction <- 1      # 0=up, 1=right, 2=down, 3=left

segment_x[0] <- 10
segment_x[1] <- 9
segment_x[2] <- 8
segment_x[3] <- 7
segment_y[0] <- 9
segment_y[1] <- 9
segment_y[2] <- 9
segment_y[3] <- 9

food <- 180         # pixel index: row 5, column 20
candidate <- 180

i <- 0
WHILE i < length DO
    address <- segment_y[i] * 32
    address <- address + segment_x[i]
    address <- address * 4
    address <- address + base_address
    memory[address] <- snake_color
    i <- i + 1
END-WHILE

address <- food * 4
address <- address + base_address
memory[address] <- food_color

WHILE TRUE DO
    WAIT 150     # 150 millisecond pause

    key <- memory[keyboard_address]

    IF key != 0 THEN
        IF key == 38 AND direction != 2 THEN
            direction <- 0
        END-IF
        IF key == 40 AND direction != 0 THEN
            direction <- 2
        END-IF
        IF key == 37 AND direction != 1 THEN
            direction <- 3
        END-IF
        IF key == 39 AND direction != 3 THEN
            direction <- 1
        END-IF
        memory[keyboard_address] <- 0
    END-IF

    new_x <- segment_x[0]
    new_y <- segment_y[0]

    IF direction == 0 THEN
        new_y <- new_y - 1
        IF new_y < 0 THEN
            new_y <- 17
        END-IF
    END-IF

    IF direction == 2 THEN
        new_y <- new_y + 1
        IF new_y >= 18 THEN
            new_y <- 0
        END-IF
    END-IF

    IF direction == 3 THEN
        new_x <- new_x - 1
        IF new_x < 0 THEN
            new_x <- 31
        END-IF
    END-IF

    IF direction == 1 THEN
        new_x <- new_x + 1
        IF new_x >= 32 THEN
            new_x <- 0
        END-IF
    END-IF

    position <- new_y * 32
    position <- position + new_x

    grew <- 0
    IF position == food THEN
        length <- length + 1
        grew <- 1
    END-IF

    limit <- length - 1

    collided <- 0
    i <- 0
    WHILE i < limit AND collided == 0 DO
        IF new_x == segment_x[i] AND new_y == segment_y[i] THEN
            collided <- 1
        END-IF
        i <- i + 1
    END-WHILE

    IF collided == 1 THEN
        STOP
    END-IF

    IF grew == 0 THEN
        address <- segment_y[length - 1] * 32
        address <- address + segment_x[length - 1]
        address <- address * 4
        address <- address + base_address
        memory[address] <- 0
    END-IF

    i <- length - 1
    WHILE i > 0 DO
        segment_x[i] <- segment_x[i - 1]
        segment_y[i] <- segment_y[i - 1]
        i <- i - 1
    END-WHILE

    segment_x[0] <- new_x
    segment_y[0] <- new_y

    address <- segment_y[0] * 32
    address <- address + segment_x[0]
    address <- address * 4
    address <- address + base_address
    memory[address] <- snake_color

    IF grew == 1 THEN
        IF length == capacity THEN
            STOP
        END-IF

        candidate <- candidate + 37
        IF candidate >= 576 THEN
            candidate <- candidate - 576
        END-IF

        index <- candidate
        address <- index * 4
        address <- address + base_address

        WHILE memory[address] != 0 DO
            index <- index + 1
            IF index >= 576 THEN
                index <- 0
            END-IF
            address <- index * 4
            address <- address + base_address
        END-WHILE

        food <- index
        memory[address] <- food_color
    END-IF
END-WHILE

END
```

## RISC-V Program

```asm
.text
main:
    lui  t0, 0xFF000              # framebuffer base = 0xFF000000
    lui  t3, 0xFF010              # keyboard register = 0xFF010000

    lui  s2, 0x10010              # s2 = 0x10010000 (base of the array of columns)
    lui  s3, 0x10011              # s3 = 0x10011000 (base of the array of rows)

    addi s4, zero, 4              # initial length
    addi s6, zero, 576            # capacity = 32 * 18 (whole panel)
    addi s1, zero, 1              # initial direction: right

    addi s7, zero, 180            # food index (row 5, column 20)
    addi s9, zero, 180            # candidate index of the next draw

    # initial body: columns 10, 9, 8 and 7, all on row 9
    addi t1, zero, 10
    sw   t1, 0(s2)
    addi t1, zero, 9
    sw   t1, 4(s2)
    addi t1, zero, 8
    sw   t1, 8(s2)
    addi t1, zero, 7
    sw   t1, 12(s2)

    addi t1, zero, 9
    sw   t1, 0(s3)
    sw   t1, 4(s3)
    sw   t1, 8(s3)
    sw   t1, 12(s3)

    addi s5, zero, 0
init_draw:
    slli t6, s5, 2
    add  t2, s2, t6
    lw   a4, 0(t2)
    add  t2, s3, t6
    lw   a5, 0(t2)
    slli t2, a5, 5
    add  t2, t2, a4
    slli t2, t2, 2
    add  t2, t2, t0
    lui  t1, 0xF                  # green, 0x0000F000
    sw   t1, 0(t2)
    addi s5, s5, 1
    blt  s5, s4, init_draw

    slli t2, s7, 2                # address of the initial food
    add  t2, t2, t0
    lui  t1, 0xFF0                # red (food)
    sw   t1, 0(t2)

tick:
    addi a0, zero, 150            # 150 ms pause per game round
    addi a7, zero, 32             # system call 32 = pause (a0 = ms)
    ecall

    lw   t4, 0(t3)
    beq  t4, zero, move_step

    addi t5, zero, 38
    beq  t4, t5, check_up
    addi t5, zero, 40
    beq  t4, t5, check_down
    addi t5, zero, 37
    beq  t4, t5, check_left
    addi t5, zero, 39
    beq  t4, t5, check_right
    jal  zero, consume_key

check_up:
    addi t5, zero, 2              # opposite of up (0) is down (2)
    beq  s1, t5, consume_key      # going down? ignores the key
    addi s1, zero, 0
    jal  zero, consume_key

check_down:
    addi t5, zero, 0              # opposite of down (2) is up (0)
    beq  s1, t5, consume_key
    addi s1, zero, 2
    jal  zero, consume_key

check_left:
    addi t5, zero, 1              # opposite of left (3) is right (1)
    beq  s1, t5, consume_key
    addi s1, zero, 3
    jal  zero, consume_key

check_right:
    addi t5, zero, 3              # opposite of right (1) is left (3)
    beq  s1, t5, consume_key
    addi s1, zero, 1

consume_key:
    sw   x0, 0(t3)

move_step:
    lw   a4, 0(s2)
    lw   a5, 0(s3)

    addi t5, zero, 0
    beq  s1, t5, move_up
    addi t5, zero, 1
    beq  s1, t5, move_right
    addi t5, zero, 2
    beq  s1, t5, move_down
    jal  zero, move_left

move_up:
    addi a0, a5, -1
    bge  a0, zero, up_ok
    addi a0, zero, 17
up_ok:
    addi a1, a0, 0
    addi a0, a4, 0
    jal  zero, head_computed

move_down:
    addi a1, a5, 1
    addi t5, zero, 18
    blt  a1, t5, down_ok
    addi a1, zero, 0
down_ok:
    addi a0, a4, 0
    jal  zero, head_computed

move_left:
    addi a0, a4, -1
    bge  a0, zero, left_ok
    addi a0, zero, 31
left_ok:
    addi a1, a5, 0
    jal  zero, head_computed

move_right:
    addi a0, a4, 1
    addi t5, zero, 32
    blt  a0, t5, right_ok
    addi a0, zero, 0
right_ok:
    addi a1, a5, 0

head_computed:
    slli t5, a1, 5
    add  t5, t5, a0               # t5 = index of the new head (y * 32 + x)
    addi a7, zero, 0              # a7 = 1 if the snake grows on this round
    bne  t5, s7, skip_grow
    addi s4, s4, 1
    addi a7, zero, 1
skip_grow:

    # segments to check: length - 1. If it grew, the last index
    # does not have a segment yet. If it did not grow, the tail frees the cell
    addi t6, s4, -1

    addi s5, zero, 0
collision_loop:
    bge  s5, t6, no_collision
    slli t5, s5, 2
    add  t2, s2, t5
    lw   a2, 0(t2)
    add  t2, s3, t5
    lw   a3, 0(t2)
    bne  a0, a2, collision_next
    bne  a1, a3, collision_next
    jal  zero, fim                # collision with its own body: ends
collision_next:
    addi s5, s5, 1
    jal  zero, collision_loop

no_collision:
    bne  a7, zero, do_shift
    addi t6, s4, -1
    slli t6, t6, 2
    add  t2, s2, t6
    lw   a2, 0(t2)
    add  t2, s3, t6
    lw   a3, 0(t2)

do_shift:
    addi s5, s4, -1
shift_loop:
    beq  s5, zero, shift_done
    slli t6, s5, 2
    addi t5, t6, -4
    add  t2, s2, t5
    lw   a4, 0(t2)
    add  t2, s3, t5
    lw   a5, 0(t2)
    add  t2, s2, t6
    sw   a4, 0(t2)
    add  t2, s3, t6
    sw   a5, 0(t2)
    addi s5, s5, -1
    jal  zero, shift_loop
shift_done:
    sw   a0, 0(s2)
    sw   a1, 0(s3)

    bne  a7, zero, draw_new_head
    slli t2, a3, 5
    add  t2, t2, a2
    slli t2, t2, 2
    add  t2, t2, t0
    sw   x0, 0(t2)

draw_new_head:
    slli t2, a1, 5
    add  t2, t2, a0
    slli t2, t2, 2
    add  t2, t2, t0
    lui  t1, 0xF                  # green, 0x0000F000
    sw   t1, 0(t2)

    beq  a7, zero, tick

spawn_food:
    beq  s4, s6, fim              # whole panel occupied: victory, ends

    addi s9, s9, 37               # step 37: coprime with 576, goes through the whole panel
    addi t5, zero, 576
    blt  s9, t5, candidate_ok
    addi s9, s9, -576             # passed the last pixel, goes back to the start
candidate_ok:
    addi t6, s9, 0

find_free:
    slli t2, t6, 2
    add  t2, t2, t0               # address of the candidate pixel
    lw   t5, 0(t2)
    beq  t5, zero, free_found     # unlit pixel: free
    addi t6, t6, 1
    addi t5, zero, 576
    blt  t6, t5, find_free
    addi t6, zero, 0              # passed the last pixel, goes back to the first one
    jal  zero, find_free

free_found:
    addi s7, t6, 0                # new food index
    lui  t1, 0xFF0                # red (food)
    sw   t1, 0(t2)
    jal  zero, tick

fim:
    addi a7, zero, 10             # system call 10 = ends the program
    ecall
```

:::note
This program uses the `ecall` instruction. The system calls supported by the simulator are listed in [Syscalls](/aeris-docs/riscv/syscalls/).
:::

## Explanation

The body is kept in two arrays, one with the columns and the other with the rows of the segments, and index 0 corresponds to the head. On each iteration, the program pauses, computes the new position of the head and checks whether it coincides with the food, in which case the body grows, or with some segment of the body, in which case the program ends. When the snake does not grow, the tail is disregarded in this comparison, since it frees its cell. The body is shifted from the tail to the head, with each segment taking the position of the previous one. The pixels are numbered from 0 to 575, so that the pixel of column x and row y receives the number y * 32 + x. The food position starts from a counter that advances 37 positions at a time, in circular counting, and, if the candidate pixel is lit, the program examines the following ones until it finds an unlit pixel. The key corresponding to the direction opposite to the current one is ignored. The game ends by collision or when the body occupies all the pixels, and the program finishes with [`ecall`](/aeris-docs/riscv/syscalls/).
